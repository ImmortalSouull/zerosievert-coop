					build UiMinimapPlayerMarker {
						coopMate = coopI
						xy = [CoopMateX(coopI), CoopMateY(coopI)]
						color = CoopMateColor(coopI)

						step = fn {
							xy = [CoopMateX(coopMate), CoopMateY(coopMate)]
						}
					}
				}
				coopI += 1
			}

				build UiCoopMark {
					xy = [CoopMarkX(coopI), CoopMarkY(coopI)]
					markColor = CoopMarkColor(coopI)
				}
